import pytest
from src.engine import liquidar_nomina


def test_cp_01_recargo_diurno_25():
    """Test CP-01: Recargo diurno 25%"""
    resultado = liquidar_nomina(2_000_000, 10, 0, vlr_hora=10_000)
    assert resultado["recargo_diurno"] == 125_000.0
    assert resultado["recargo_nocturno"] == 0.0
    assert resultado["total_devengado"] == 2_125_000.0


def test_cp_02_recargo_nocturno_75():
    """Test CP-02: Recargo nocturno 75%"""
    resultado = liquidar_nomina(2_000_000, 0, 8, vlr_hora=12_000)
    assert resultado["recargo_nocturno"] == 168_000.0
    assert resultado["recargo_diurno"] == 0.0
    assert resultado["total_devengado"] == 2_168_000.0


def test_cp_03_recargos_simultaneos():
    """Test CP-03: Recargos diurno y nocturno simultáneos"""
    resultado = liquidar_nomina(3_000_000, 5, 4, vlr_hora=15_000)
    assert resultado["recargo_diurno"] == 93_750.0
    assert resultado["recargo_nocturno"] == 105_000.0
    assert resultado["total_devengado"] == 3_198_750.0


def test_cp_04_deducciones_sobre_total_devengado():
    """Test CP-04: Deducciones sobre total devengado"""
    resultado = liquidar_nomina(2_500_000, 5, 2, vlr_hora=10_000)
    assert resultado["total_devengado"] == 2_597_500.0
    assert resultado["descuento_salud"] == 103_900.0
    assert resultado["descuento_pension"] == 103_900.0


def test_cp_05_auxilio_transporte_aplica():
    """Test CP-05: Auxilio de transporte aplica (salario en el tope)"""
    resultado = liquidar_nomina(2_600_000, 0, 0, vlr_hora=10_000)
    assert resultado["auxilio_transporte"] == 162_000.0


def test_cp_06_auxilio_transporte_no_aplica():

    resultado = liquidar_nomina(2_600_001, 0, 0, vlr_hora=10_000)
    assert resultado["auxilio_transporte"] == 0.0

def test_cp_10_calculo_completo():
    """Test CP-10: Cálculo completo de nómina"""

    assert resultado["recargo_nocturno"] == 52_500.0
    assert resultado["total_devengado"] == 2_527_500.0
    assert resultado["descuento_salud"] == 101_100.0
    assert resultado["descuento_pension"] == 101_100.0
