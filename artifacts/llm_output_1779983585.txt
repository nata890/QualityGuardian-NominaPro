import pytest
from src.engine import liquidar_nomina

# ------------------------------------------------------------------------
# CP-01 – Recargo diurno 25%
# ------------------------------------------------------------------------
def test_cp_01():
    """CP-01: Recargo diurno 25%."""
    resultado = liquidar_nomina(2_000_000, 10, 0, 10_000)
    assert resultado["recargo_diurno"] == 125_000.0
    assert resultado["recargo_nocturno"] == 0.0
    assert resultado["total_devengado"] == 2_125_000.0

# ------------------------------------------------------------------------
# CP-02 – Recargo nocturno 75%
# ------------------------------------------------------------------------
def test_cp_02():
    """CP-02: Recargo nocturno 75%."""
    resultado = liquidar_nomina(2_000_000, 0, 8, 12_000)
    assert resultado["recargo_nocturno"] == 168_000.0
    assert resultado["recargo_diurno"] == 0.0
    assert resultado["total_devengado"] == 2_168_000.0

# ------------------------------------------------------------------------
# CP-03 – Recargos diurno y nocturno simultáneos
# ------------------------------------------------------------------------
def test_cp_03():
    """CP-03: Recargos diurno y nocturno simultáneos."""
    resultado = liquidar_nomina(3_000_000, 5, 4, 15_000)
    assert resultado["recargo_diurno"] == 93_750.0
    assert resultado["recargo_nocturno"] == 105_000.0
    assert resultado["total_devengado"] == 3_198_750.0

# ------------------------------------------------------------------------
# CP-04 – Deducciones sobre total devengado
# ------------------------------------------------------------------------
def test_cp_04():
    """CP-04: Deducciones sobre total devengado."""
    resultado = liquidar_nomina(2_500_000, 5, 2, 10_000)
    assert resultado["total_devengado"] == 2_597_500.0
    assert resultado["descuento_salud"] == 103_900.0
    assert resultado["descuento_pension"] == 103_900.0

# ------------------------------------------------------------------------
# CP-05 – Auxilio de transporte aplica (salario en el tope)
# ------------------------------------------------------------------------
def test_cp_05():
    """CP-05: Auxilio de transporte aplica (salario en el tope)."""
    resultado = liquidar_nomina(2_600_000, 0, 0, 10_000)
    assert resultado["auxilio_transporte"] == 162_000.0

# ------------------------------------------------------------------------
# CP-06 – Auxilio de transporte no aplica (salario sobre el tope)
# ------------------------------------------------------------------------
def test_cp_06():

        liquidar_nomina(500_000, 0, 0, 10_000)

# ------------------------------------------------------------------------
# CP-08 – Horas extras diurnas negativas lanza ValueError
# ------------------------------------------------------------------------
def test_cp_08():

        liquidar_nomina(2_000_000, -3, 0, 10_000)

# ------------------------------------------------------------------------
# CP-09 – Horas extras nocturnas negativas lanza ValueError
# ------------------------------------------------------------------------
def test_cp_09():
    """CP-09: Horas extras nocturnas negativas lanza ValueError."""
    with pytest.raises(ValueError, match=r"Las horas extras nocturnas \(-2\) no pueden ser negativas\."):

# ------------------------------------------------------------------------
# CP-10 – Cálculo completo de nómina
# ------------------------------------------------------------------------

    assert resultado["descuento_salud"] == 101_100.0
