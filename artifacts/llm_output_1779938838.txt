import pytest

def test_cp01_recargo_diurno_25():
    """
    CP-01: Recargo diurno 25%
    - Salario base: 2.000.000
    - HE diurnas: 10
    - HE nocturnas: 0
    - Valor hora: 10.000
    -> Recargo diurno = 10 * 10.000 * 1.25 = 125.000,0
    -> Total devengado = 2.125.000,0
    """
    resultado = liquidar_nomina(
        salario_base=2_000_000,
        horas_extras_diurnas=10,
        horas_extras_nocturnas=0,
        valor_hora=10_000
    )
    assert resultado["recargo_diurno"] == 125_000.0
    assert resultado["recargo_nocturno"] == 0.0
    assert resultado["total_devengado"] == 2_125_000.0

def test_cp04_deducciones_sobre_devengado():
    """
    CP-04: Deducciones sobre total devengado
    - Salario base: 2.500.000
    - HE diurnas: 5
    - HE nocturnas: 2
    - Valor hora: 10.000
    -> Total devengado = 2.597.500,0
    -> Descuento salud 4% = 103.900,0
    -> Descuento pensión 4% = 103.900,0
    """
    resultado = liquidar_nomina(
        salario_base=2_500_000,
        horas_extras_diurnas=5,
        horas_extras_nocturnas=2,
        valor_hora=10_000
    )
    total_dev = resultado["total_devengado"]
    assert total_dev == 2_597_500.0
    assert resultado["descuento_salud"] == 103_900.0
    assert resultado["descuento_pension"] == 103_900.0
    # Verificar que se calculan sobre total devengado, no solo base
    assert resultado["descuento_salud"] == round(total_dev * 0.04, 2)
    assert resultado["descuento_pension"] == round(total_dev * 0.04, 2)
